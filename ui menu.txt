-- Хранилище добавленных функций
local functions = {}

-- Создание элемента функции
local function createFunction(name, callback)
    local button = Instance.new("TextButton")
    button.Name = name
    button.Size = UDim2.new(1, -4, 0, 48)
    button.BackgroundColor3 = Color3.fromRGB(29, 29, 36)
    button.BorderSizePixel = 0
    button.Text = name
    button.TextColor3 = Color3.fromRGB(230, 230, 235)
    button.TextSize = 15
    button.Font = Enum.Font.GothamMedium
    button.TextXAlignment = Enum.TextXAlignment.Left
    button.AutoButtonColor = false
    button.Parent = list

    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, 10)
    corner.Parent = button

    local buttonPadding = Instance.new("UIPadding")
    buttonPadding.PaddingLeft = UDim.new(0, 15)
    buttonPadding.Parent = button

    local line = Instance.new("Frame")
    line.Name = "Indicator"
    line.Size = UDim2.fromOffset(3, 24)
    line.Position = UDim2.new(1, -8, 0.5, -12)
    line.BackgroundColor3 = Color3.fromRGB(130, 130, 150)
    line.BorderSizePixel = 0
    line.Parent = button

    local lineCorner = Instance.new("UICorner")
    lineCorner.CornerRadius = UDim.new(1, 0)
    lineCorner.Parent = line

    button.MouseEnter:Connect(function()
        button.BackgroundColor3 = Color3.fromRGB(40, 40, 50)
    end)

    button.MouseLeave:Connect(function()
        button.BackgroundColor3 = Color3.fromRGB(29, 29, 36)
    end)

    button.MouseButton1Click:Connect(function()
        if typeof(callback) == "function" then
            callback()
        end
    end)

    return button
end

-- Добавление функции
function AddFunction(name, callback)
    if not name then
        return
    end

    if functions[name] then
        functions[name]:Destroy()
        functions[name] = nil
    end

    functions[name] = createFunction(name, callback)
end

-- Удаление функции
function RemoveFunction(name)
    if functions[name] then
        functions[name]:Destroy()
        functions[name] = nil
    end
end

-- Полная очистка функций
function ClearFunctions()
    for name, object in pairs(functions) do
        if object then
            object:Destroy()
        end
        functions[name] = nil
    end
end

-- Перетаскивание меню
local dragging = false
local dragStart
local startPosition

topBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then

        dragging = true
        dragStart = input.Position
        startPosition = main.Position

        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if not dragging then
        return
    end

    if input.UserInputType ~= Enum.UserInputType.MouseMovement
        and input.UserInputType ~= Enum.UserInputType.Touch then
        return
    end

    local delta = input.Position - dragStart

    main.Position = UDim2.new(
        startPosition.X.Scale,
        startPosition.X.Offset + delta.X,
        startPosition.Y.Scale,
        startPosition.Y.Offset + delta.Y
    )
end)

    main.Position = piButton.Position
    piButton.Visible = false
    main.Visible = true
end)

-- Перетаскивание π-кнопки
local piDragging = false
local piDragStart
local piStartPosition

piButton.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then

        piDragging = true
        piDragStart = input.Position
        piStartPosition = piButton.Position

        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                piDragging = false
            end
        end)
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if not piDragging then
        return
    end

    if input.UserInputType ~= Enum.UserInputType.MouseMovement
        and input.UserInputType ~= Enum.UserInputType.Touch then
        return
    end

    local delta = input.Position - piDragStart

    piButton.Position = UDim2.new(
        piStartPosition.X.Scale,
        piStartPosition.X.Offset + delta.X,
        piStartPosition.Y.Scale,
        piStartPosition.Y.Offset + delta.Y
    )
end
