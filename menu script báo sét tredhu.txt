        if typeof(callback) == "function" then
            callback(button, line)

    end)

function AddFunction(name, callback)

    if not name then
        return
    end

    if functions[name] then
        functions[name]:Destroy()
        functions[name] = nil
    end

    functions[name] =
        createFunction(name, callback)
end

function RemoveFunction(name)

    if functions[name] then
        functions[name]:Destroy()
        functions[name] = nil
    end
end

function ClearFunctions()

    for name, object in pairs(functions) do

        if object then
            object:Destroy()
        end

        functions[name] = nil
    end
end

local function LightningWarning()

    if not LightningEnabled then
        return
    end

    if CountdownRunning then
        return
    end

    local now = os.clock()

    if now - LastAlert < Cooldown then
        return
    end

    LastAlert = now
    CountdownRunning = true

    WarningToken += 1

    local token =
        WarningToken

    Indicator.Visible = true
    Alert.Visible = true

    Indicator.BackgroundColor3 =
        Color3.fromRGB(255, 210, 0)

    AlertSound:Stop()
    AlertSound:Play()

    for i = 3, 1, -1 do

        if not LightningEnabled
            or token ~= WarningToken then

            break
        end

        Alert.Text =
            "⚡ SÉT SẮP ĐÁNH\nHÁI CÂY: " .. i

        task.wait(1)
    end

    if LightningEnabled
        and token == WarningToken then

        Alert.Text =
            "⚡ SÉT ĐÁNH!\n🌳 HÁI NGAY!"

        Indicator.BackgroundColor3 =
            Color3.fromRGB(255, 255, 255)

        AlertSound:Stop()
        AlertSound:Play()

        task.wait(1.2)
    end

    if token == WarningToken then

        Alert.Visible = false
        Indicator.Visible = false

        Indicator.BackgroundColor3 =
            Color3.fromRGB(255, 210, 0)

        CountdownRunning = false
    end
end

local function IsLightningEffect(Object)

    return Object:IsA("Highlight")
        or Object:IsA("SelectionBox")
        or Object:IsA("Beam")
        or Object:IsA("ParticleEmitter")
end

Workspace.DescendantAdded:Connect(function(Object)

    if LightningEnabled
        and IsLightningEffect(Object) then

        task.delay(0.05, function()

            if LightningEnabled then
                LightningWarning()
            end
        end)
    end

    -- Anti Lag xử lý effect mới
    if AntiLagEnabled then

        task.defer(function()

            if not Object
                or not Object.Parent then
                return
            end

            if Object:IsA("ParticleEmitter") then

                SavedEffects[Object] =
                    SavedEffects[Object] or {}

                if SavedEffects[Object].Enabled == nil then
                    SavedEffects[Object].Enabled =
                        Object.Enabled
                end

                Object.Enabled = false

                if SavedEffects[Object].Rate == nil then
                    SavedEffects[Object].Rate =
                        Object.Rate
                end

                Object.Rate = 0

            elseif Object:IsA("Trail")
                or Object:IsA("Smoke")
                or Object:IsA("Fire")
                or Object:IsA("Sparkles")
                or Object:IsA("PostEffect") then

                SavedEffects[Object] =
                    SavedEffects[Object] or {}

                if SavedEffects[Object].Enabled == nil then
                    SavedEffects[Object].Enabled =
                        Object.Enabled
                end

                Object.Enabled = false
            end
        end)
    end
end)

local LightningKeywords = {
    "lightning",
    "thunder",
    "strike",
    "storm",
    "electric",
    "electricity",
    "thunderbolt",
    "thunderstorm",
    "lightningbolt",
    "bolt",
    "zap"
}

local function IsLightningSound(Name)

    Name = string.lower(Name)

    for _, Keyword in ipairs(LightningKeywords) do

        if string.find(
            Name,
            Keyword,
            1,
            true
        ) then

            return true
        end
    end

    return false
end

local function MonitorSound(Object)

    if not Object:IsA("Sound") then
        return
    end

    if not IsLightningSound(Object.Name) then
        return
    end

    Object.Played:Connect(function()

        if LightningEnabled then
            LightningWarning()
        end
    end)
end

for _, Object in ipairs(
    Workspace:GetDescendants()
) do

    MonitorSound(Object)
end

Workspace.DescendantAdded:Connect(
    MonitorSound
)

local function EnableAntiLag()

    -- Giảm shadow
    if SavedLighting.GlobalShadows == nil then

        SavedLighting.GlobalShadows =
            Lighting.GlobalShadows
    end

    Lighting.GlobalShadows = false

    -- Giảm lighting quality
    pcall(function()

        if SavedLighting.EnvironmentDiffuseScale == nil then
            SavedLighting.EnvironmentDiffuseScale =
                Lighting.EnvironmentDiffuseScale
        end

        Lighting.EnvironmentDiffuseScale = 0

    end)

    pcall(function()

        if SavedLighting.EnvironmentSpecularScale == nil then
            SavedLighting.EnvironmentSpecularScale =
                Lighting.EnvironmentSpecularScale
        end

        Lighting.EnvironmentSpecularScale = 0

    end)

    -- Tắt post effect nặng
    for _, Object in ipairs(
        Lighting:GetChildren()
    ) do

        if Object:IsA("PostEffect") then

            if SavedEffects[Object] == nil then

                SavedEffects[Object] = {
                    Enabled = Object.Enabled
                }
            end

            Object.Enabled = false
        end
    end

    -- Tối ưu effect trong Workspace
    for _, Object in ipairs(
        Workspace:GetDescendants()
    ) do

        if Object:IsA("ParticleEmitter") then

            SavedEffects[Object] =
                SavedEffects[Object] or {}

            if SavedEffects[Object].Enabled == nil then
                SavedEffects[Object].Enabled =
                    Object.Enabled
            end

            if SavedEffects[Object].Rate == nil then
                SavedEffects[Object].Rate =
                    Object.Rate
            end

            Object.Enabled = false
            Object.Rate = 0

        elseif Object:IsA("Trail")
            or Object:IsA("Smoke")
            or Object:IsA("Fire")
            or Object:IsA("Sparkles") then

            SavedEffects[Object] =
                SavedEffects[Object] or {}

            if SavedEffects[Object].Enabled == nil then
                SavedEffects[Object].Enabled =
                    Object.Enabled
            end

            Object.Enabled = false
        end
    end

    -- Blur 9: chỉ ảnh hưởng phần 3D,
    -- GUI vẫn rõ để đọc menu/cảnh báo
    if not BlurEffect then

        BlurEffect =
            Instance.new("BlurEffect")

        BlurEffect.Name =
            "TredhuGoldSrcBlur"

        BlurEffect.Size = 9
        BlurEffect.Parent = Lighting
    end
end
