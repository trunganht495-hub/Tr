PowerButton.Name = "PowerButton"
PowerButton.Parent = SpeedGui
PowerButton.BackgroundColor3 = Color3.fromRGB(60, 60, 60)
PowerButton.BorderSizePixel = 0
PowerButton.Position = UDim2.new(0.1, 0, 0.7, 0)
PowerButton.Size = UDim2.new(0.8, 0, 0, 35)
PowerButton.Font = Enum.Font.SourceSansBold
PowerButton.Text = "Bật / Tắt: TẮT"
PowerButton.TextColor3 = Color3.fromRGB(255, 50, 50)
PowerButton.TextSize = 14

Menu1Button.MouseButton1Click:Connect(function()
	SpeedGui.Visible = not SpeedGui.Visible
end)

RunService.Heartbeat:Connect(function()
	if speedEnabled then
		local character = LocalPlayer.Character
		if character and character:FindFirstChild("Humanoid") then
			character.Humanoid.WalkSpeed = currentSpeed
		end
	end
end)

PowerButton.MouseButton1Click:Connect(function()
	speedEnabled = not speedEnabled
	if speedEnabled then
		PowerButton.Text = "Bật / Tắt: BẬT"
		PowerButton.TextColor3 = Color3.fromRGB(50, 255, 50)
	else
		PowerButton.Text = "Bật / Tắt: TẮT"
		PowerButton.TextColor3 = Color3.fromRGB(255, 50, 50)
		local character = LocalPlayer.Character
		if character and character:FindFirstChild("Humanoid") then
			character.Humanoid.WalkSpeed = 16
		end
	end
end)

DecButton.MouseButton1Click:Connect(function()
	currentSpeed = currentSpeed - 2
	SpeedLabel.Text = tostring(currentSpeed)
end)

IncButton.MouseButton1Click:Connect(function()
	currentSpeed = currentSpeed + 2
	SpeedLabel.Text = tostring(currentSpeed)
end)
